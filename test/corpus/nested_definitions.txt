module Nested {
  def double(x) =
    def plus(a, b) = a + b
    plus(x, x)

  def pow4(x) =
    val x2 = x * x
    x2 * x2

  def triple(x) =
    def add(n) = n + x; add(x, add(x, x))
}

module NestedTemporal {
  var x: int

  temporal property =
    temporal A = eventually(x > 3)
    temporal B = always(eventually(x == 0))
    A implies B
}

module NestedAction {
  var n: int

  action step =
    nondet i = oneOf(0.to(10))
    action update = n' = i
    all {
      i > n,
      update,
    }
}
