module UfcsTemporal {
  var n: int
  temporal invariant = (n >= 0).always
  temporal progress = (n == 1).eventually
  temporal nextInvariant = (n >= 0).next
}

module EnabledAction {
  var n: int
  action increment = n' = n + 1
  temporal canIncrement = increment.enabled
}

Action temporal conversion
========================================

module ActionConversion {
  var n: int
  action increment = n' = n + 1
  temporal mayStutter = increment.orKeep(n)
  temporal mustAdvance = increment.mustChange(n)
}
